# --- Servidor -----------------------------------------------------------
PORT=3000

# Cuántos proxies inversos hay delante de este servidor cuando lo
# despliegues (Render/Railway/Fly/nginx/Cloudflare suelen ser 1). En tu
# máquina local esto no importa. Solo lo tocas si te aparece un aviso de
# "trust proxy" de express-rate-limit en producción.
TRUST_PROXY_HOPS=1

# URL exacta de tu sitio en producción (sin barra al final), ej:
#   ALLOWED_ORIGIN=https://miusuario.github.io
# Se usa para dos cosas: (1) restringir CORS para que solo tu propio
# sitio pueda llamar a esta API, y (2) evitar que /api/send-reset-email
# se use para mandar enlaces que no sean del propio sitio. Déjalo vacío
# mientras pruebas en local (file:// o localhost) — solo complétalo
# cuando publiques el sitio de verdad.
ALLOWED_ORIGIN=

# --- Asistente de IA (Anthropic / Claude) --------------------------------
# Crea tu llave en https://console.anthropic.com/settings/keys — necesitas
# una cuenta de Anthropic con facturación activada (la API es de pago por
# uso; el consumo de un sitio de estudio personal es muy bajo).
ANTHROPIC_API_KEY=

# Modelo a usar. claude-sonnet-5 es un buen punto medio de calidad/costo
# para este caso (responder preguntas + decidir cuándo buscar). Si quieres
# recortar costos, puedes cambiar a claude-haiku-4-5-20251001 (más rápido
# y barato); si prefieres la mejor calidad posible, claude-opus-4-8.
ANTHROPIC_MODEL=claude-sonnet-5

# Tope de tokens de salida por respuesta del asistente.
ANTHROPIC_MAX_TOKENS=1536

# --- Correo (SMTP) para recuperación de contraseña -----------------------
# Sirve con cualquier proveedor SMTP. Ejemplo con Gmail (requiere activar
# la verificación en 2 pasos y crear una "Contraseña de aplicación" en
# https://myaccount.google.com/apppasswords — la contraseña normal de
# Gmail NO funciona aquí):
#   SMTP_HOST=smtp.gmail.com
#   SMTP_PORT=587
#   SMTP_SECURE=false
#   SMTP_USER=tucuenta@gmail.com
#   SMTP_PASS=la-contraseña-de-aplicación-de-16-letras
#   EMAIL_FROM=Herbario PAES <tucuenta@gmail.com>
#
# Alternativas con plan gratuito para volumen bajo (más confiables que
# Gmail para esto, y pensadas justo para correo transaccional): Resend
# (resend.com), Brevo (brevo.com), Zoho ZeptoMail. Todas dan un host/
# puerto/usuario/contraseña SMTP que van directo en estas mismas 5
# variables — no hace falta cambiar código, solo estos valores.
SMTP_HOST=
SMTP_PORT=587
SMTP_SECURE=false
SMTP_USER=
SMTP_PASS=
EMAIL_FROM=

# --- Límites de tasa (opcional, valores por defecto ya son razonables) ---
# Máximo de correos de recuperación por IP cada 15 minutos.
RATE_LIMIT_EMAIL_MAX=5
# Máximo de mensajes al chat de IA por IP cada 10 minutos.
RATE_LIMIT_CHAT_MAX=20
